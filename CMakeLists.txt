cmake_minimum_required(VERSION 3.25)
project(PlanetLab VERSION 0.15.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(PLANET_BUILD_DESKTOP "Build the Win32 explorer" ${WIN32})
option(PLANET_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

add_library(planet_core src/model.cpp src/landforms.cpp src/surface_detail.cpp src/flora.cpp src/navigation.cpp src/weather.cpp src/global_weather.cpp src/hydrology.cpp src/water_motion.cpp src/sound.cpp src/preferences.cpp src/globe_camera.cpp src/lighting.cpp src/shadows.cpp src/detail_mesh.cpp src/rivers.cpp src/terrain_mesh.cpp)
find_package(Threads REQUIRED)
target_link_libraries(planet_core PUBLIC Threads::Threads)
target_include_directories(planet_core PUBLIC include)
if(MSVC)
    target_compile_options(planet_core PUBLIC /W4 /permissive- /utf-8 /EHsc)
else()
    target_compile_options(planet_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    if(PLANET_SANITIZERS)
        target_compile_options(planet_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(planet_core PUBLIC -fsanitize=address,undefined)
    endif()
endif()

add_executable(planet_cli src/cli.cpp)
target_link_libraries(planet_cli PRIVATE planet_core)

if(PLANET_BUILD_DESKTOP)
    if(NOT WIN32)
        message(FATAL_ERROR "The explorer is a native Win32 app. Build planet_cli on other platforms.")
    endif()
    add_executable(PlanetLab WIN32 src/win32_main.cpp src/renderer.cpp src/win32_audio.cpp src/win32_dialogs.cpp)
    target_link_libraries(PlanetLab PRIVATE planet_core opengl32 gdi32 user32 shell32 winmm comctl32)
    target_compile_definitions(PlanetLab PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MINGW)
        target_link_options(PlanetLab PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    install(TARGETS PlanetLab planet_cli RUNTIME DESTINATION .)
    install(FILES README.md DESTINATION .)
    install(FILES docs/SCIENCE.md docs/VALIDATION.md docs/PERFORMANCE.md DESTINATION docs)
endif()
if(MINGW)
    target_link_options(planet_cli PRIVATE -static -static-libgcc -static-libstdc++)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(planet_tests tests/core_tests.cpp)
    target_link_libraries(planet_tests PRIVATE planet_core)
    if(MINGW)
        target_link_options(planet_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    foreach(suite IN ITEMS astronomy atmosphere terrain modules navigation climate detail weather hydrology landforms water_motion flora global_weather sound preferences globe_camera lighting shadows rivers)
        add_test(NAME ${suite} COMMAND planet_tests ${suite})
    endforeach()
    add_test(NAME cli_sample COMMAND ${CMAKE_COMMAND}
        -DCLI=$<TARGET_FILE:planet_cli> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_cli.cmake)
endif()

set(CPACK_PACKAGE_NAME PlanetLab)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR ZIP)
include(CPack)
